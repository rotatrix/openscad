cmake_minimum_required(VERSION 3.24)
project(OpenAxisCameraChecks LANGUAGES CXX)
include(FetchContent)
FetchContent_Declare(eigen URL https://gitlab.com/libeigen/eigen/-/archive/3.4.0/eigen-3.4.0.tar.gz
  URL_HASH SHA256=8586084f71f9bde545ee7fa6d00288b264a2b7ac3607b974e54d13e7162c1c72
  SOURCE_SUBDIR unused)
FetchContent_MakeAvailable(eigen)
if(NOT OPENAXIS_SOURCE_DIR)
  message(FATAL_ERROR "Set OPENAXIS_SOURCE_DIR to the SDK checkout")
endif()
set(OPENAXIS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(OPENAXIS_BUILD_DEMO OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPENAXIS_SOURCE_DIR}/cpp" openaxis)
add_executable(OpenAxisCameraTests camera.cc)
target_compile_features(OpenAxisCameraTests PRIVATE cxx_std_17)
target_include_directories(OpenAxisCameraTests PRIVATE ../../src "${eigen_SOURCE_DIR}" "${OPENAXIS_SOURCE_DIR}/cpp/include")
target_link_libraries(OpenAxisCameraTests PRIVATE OpenAxis::openaxis)
enable_testing()
add_test(NAME openaxis_camera COMMAND OpenAxisCameraTests)

# Optional native raster checks without compiling the entire application.
option(ENABLE_OVERLAY_TESTS "Build the Qt raster overlay checks" OFF)
if(ENABLE_OVERLAY_TESTS)
  find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
  find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
  add_executable(OpenAxisOverlayTests overlay.cc)
  target_include_directories(OpenAxisOverlayTests PRIVATE ../../src)
  target_link_libraries(OpenAxisOverlayTests PRIVATE OpenAxis::openaxis Qt${QT_VERSION_MAJOR}::Widgets)
  add_test(NAME openaxis_overlay COMMAND OpenAxisOverlayTests)
  set_tests_properties(openaxis_overlay PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
